cmake_minimum_required(VERSION 3.16)
project(bubbleoxy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(bubbleoxy INTERFACE)
target_include_directories(bubbleoxy INTERFACE src)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(bubbleoxy INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(simulator examples/simulator.cpp)
target_link_libraries(simulator PRIVATE bubbleoxy)
enable_testing()
add_executable(controller_tests tests/controller_tests.cpp)
target_link_libraries(controller_tests PRIVATE bubbleoxy)
add_test(NAME controller_tests COMMAND controller_tests)
